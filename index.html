✅ Código Optimizado & Automatizado (Clean Code & Safe)

from typing import Optional, Dict, Any
import sqlite3
import logging

def get_user_data(user_id: int) -> Optional[Dict[str, Any]]:
    """Obtiene datos de usuario usando consultas preparadas seguras."""
    try:
        with sqlite3.connect('app.db') as conn:
            cursor = conn.cursor()
            query = "SELECT id, name, email FROM users WHERE id = ?"
            cursor.execute(query, (user_id,))
            row = cursor.fetchone()
            return {"id": row[0], "name": row[1], "email": row[2]} if row else None
    except sqlite3.Error as err:
        logging.error(f"Database error for user {user_id}: {err}")
        return None
